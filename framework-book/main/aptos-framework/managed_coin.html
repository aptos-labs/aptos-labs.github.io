Resource Capabilities

Capabilities resource storing mint and burn capabilities. The resource is stored on the account that initialized coin CoinType.

struct Capabilities<CoinType> has key
Fields
burn_cap: coin::BurnCapability<CoinType>
freeze_cap: coin::FreezeCapability<CoinType>
mint_cap: coin::MintCapability<CoinType>

Constants

Account has no capabilities (burn/mint).

const ENO_CAPABILITIES: u64 = 1;

Function burn

Withdraw an amount of coin CoinType from account and burn it.

public entry fun burn<CoinType>(account: &signer, amount: u64)
Implementation 
public entry fun burn<CoinType>(
    account: &signer,
    amount: u64,
) acquires Capabilities {
    let account_addr = signer::address_of(account);

    assert!(
        exists<Capabilities<CoinType>>(account_addr),
        error::not_found(ENO_CAPABILITIES),
    );

    let capabilities = borrow_global<Capabilities<CoinType>>(account_addr);

    let to_burn = coin::withdraw<CoinType>(account, amount);
    coin::burn(to_burn, &capabilities.burn_cap);
}

Function initialize

Initialize new coin CoinType in Aptos Blockchain. Mint and Burn Capabilities will be stored under account in Capabilities resource.

public entry fun initialize<CoinType>(account: &signer, name: vector<u8>, symbol: vector<u8>, decimals: u8, monitor_supply: bool)
Implementation 
public entry fun initialize<CoinType>(
    account: &signer,
    name: vector<u8>,
    symbol: vector<u8>,
    decimals: u8,
    monitor_supply: bool,
) {
    let (burn_cap, freeze_cap, mint_cap) = coin::initialize<CoinType>(
        account,
        string::utf8(name),
        string::utf8(symbol),
        decimals,
        monitor_supply,
    );

    move_to(account, Capabilities<CoinType> {
        burn_cap,
        freeze_cap,
        mint_cap,
    });
}

    dst_addr: address,
    amount: u64,

    assert!(
        exists<Capabilities<CoinType>>(account_addr),

Function destroy_caps

Destroys capabilities from the account, so that the user no longer has access to mint or burn.

public entry fun destroy_caps<CoinType>(account: &signer)
Implementation 
public entry fun destroy_caps<CoinType>(account: &signer) acquires Capabilities {
    let (burn_cap, freeze_cap, mint_cap) = remove_caps<CoinType>(account);
    destroy_burn_cap(burn_cap);
    destroy_freeze_cap(freeze_cap);
    destroy_mint_cap(mint_cap);
}

Function remove_caps

Removes capabilities from the account to be stored or destroyed elsewhere

public fun remove_caps<CoinType>(account: &signer): (coin::BurnCapability<CoinType>, coin::FreezeCapability<CoinType>, coin::MintCapability<CoinType>)
Implementation 
public fun remove_caps<CoinType>(
    account: &signer
): (BurnCapability<CoinType>, FreezeCapability<CoinType>, MintCapability<CoinType>) acquires Capabilities {
    let account_addr = signer::address_of(account);
    assert!(
        exists<Capabilities<CoinType>>(account_addr),
        error::not_found(ENO_CAPABILITIES),
    );

    let Capabilities<CoinType> {
        burn_cap,
        freeze_cap,
        mint_cap,
    } = move_from<Capabilities<CoinType>>(account_addr);
    (burn_cap, freeze_cap, mint_cap)
}

Specification

High-level Requirements

No.RequirementCriticalityImplementationEnforcement
1 The initializing account should hold the capabilities to operate the coin. Critical The capabilities are stored under the initializing account under the Capabilities resource, which is distinct for a distinct type of coin. Enforced via initialize.
2 A new coin should be properly initialized. High In the initialize function, a new coin is initialized via the coin module with the specified properties. Enforced via initialize_internal.
3 Minting/Burning should only be done by the account who hold the valid capabilities. High The mint and burn capabilities are moved under the initializing account and retrieved, while minting/burning Enforced via: initialize, burn, mint.
4 If the total supply of coins is being monitored, burn and mint operations will appropriately adjust the total supply. High The coin::burn and coin::mint functions, when tracking the supply, adjusts the total coin supply accordingly. Formally verified via burn, mint and mint that burn and mint adjust the supply of the paired fungible asset by the amount (when the withdrawal or deposit runs without a hook), while the tracked coin supply and the value held in coins are unchanged (coin::CoinSupplyUnchanged). Formally verified via mint and burn that, for a tracked coin supply and an existing pairing, the coin supply plus the paired asset supply grows by the minted amount when the deposit runs without a deposit hook, and drops by the burned amount when the withdrawal runs without a withdraw hook from a store of the paired asset.
5 Before burning coins, exact amount of coins are withdrawn. High The burn function withdraws amount via coin::withdraw from the paired primary store of the signer and burns it; when the withdrawal runs without a hook, the store balance drops by exactly amount. Enforced via burn.
6 Minted coins are deposited to the provided destination address. High After the coins are minted via coin::mint they are deposited into the coinstore of the destination address. Enforced via mint.

Function burn 

public entry fun burn<CoinType>(account: &signer, amount: u64)

Withdraws amount from the paired primary store of the signer and burns it.

let account_addr = signer::address_of(account);
let paired = coin::spec_paired_metadata(type_info::type_of<CoinType>());
let store_addr = coin::spec_paired_store_address(account_addr, paired);
let existing = coin::spec_has_paired_store(account_addr, paired);
let plain = existing && fungible_asset::spec_store_no_withdraw_hook(store_addr);
// This enforces high-level requirement 3:
aborts_if !exists<Capabilities<CoinType>>(account_addr);
aborts_if !exists<coin::CoinConversionMap>(@aptos_framework);
aborts_if existing && global<fungible_asset::FungibleStore>(store_addr).frozen;
aborts_if plain && fungible_asset::spec_balance(store_addr) < amount;
// This enforces high-level requirement 5:
ensures plain ==>
    fungible_asset::spec_balance(store_addr) == old(fungible_asset::spec_balance(store_addr)) - amount;
let fa_addr = global<fungible_asset::FungibleStore>(store_addr).metadata.inner;
include coin::CoinSupplyUnchanged<CoinType>;
// This enforces high-level requirement 4:
ensures plain ==> fungible_asset::spec_supply(fa_addr) == old(fungible_asset::spec_supply(fa_addr)) - amount;
// This enforces high-level requirement 4:
include plain && fa_addr == option::borrow(paired).inner ==>
    TotalSupplyChange<CoinType> { added: 0, removed: amount };

For a tracked coin supply and an existing pairing, the coin supply plus the paired asset supply changes by added - removed.

schema TotalSupplyChange<CoinType> {
    added: num;
    removed: num;
    let coin_type = type_info::type_of<CoinType>();
    let tracked = option::is_some(global<coin::CoinInfo<CoinType>>(coin_type.account_address).supply)
        && option::is_some(coin::spec_paired_metadata(coin_type));
    ensures tracked ==> coin::spec_total_supply<CoinType>(coin_type)
        == old(coin::spec_total_supply<CoinType>(coin_type)) + added - removed;
}

Function mint 

public entry fun mint<CoinType>(account: &signer, dst_addr: address, amount: u64)

The Capabilities should exist in the signer address. The paired primary store of dst_addr should not be frozen.

let account_addr = signer::address_of(account);
// This enforces high-level requirement 3:
aborts_if !exists<Capabilities<CoinType>>(account_addr);
let addr = type_info::type_of<CoinType>().account_address;
aborts_if (amount != 0) && !exists<coin::CoinInfo<CoinType>>(addr);
let paired = coin::spec_paired_metadata(type_info::type_of<CoinType>());
let store_addr = coin::spec_paired_store_address(dst_addr, paired);
aborts_if !exists<coin::CoinConversionMap>(@aptos_framework);
aborts_if coin::spec_has_paired_store(dst_addr, paired)
    && global<fungible_asset::FungibleStore>(store_addr).frozen;
// This enforces high-level requirement 6:
include coin::PairedDepositBalance<CoinType> { owner: dst_addr, amount };
include coin::CoinSupplyUnchanged<CoinType>;
// This enforces high-level requirement 4:
let no_hook = coin::spec_paired_deposit_no_hook(dst_addr, paired);
ensures no_hook ==> fungible_asset::spec_supply(option::borrow(paired).inner)
    == old(fungible_asset::spec_supply(option::borrow(paired).inner)) + amount;
// This enforces high-level requirement 4:
include no_hook ==> TotalSupplyChange<CoinType> { added: amount, removed: 0 };
let post post_fa_addr = option::borrow(coin::spec_paired_metadata(type_info::type_of<CoinType>())).inner;
// This enforces high-level requirement 4:
ensures option::is_none(paired)
    && aptos_framework::primary_fungible_store::spec_metadata_no_deposit_hook(post_fa_addr)
    ==> fungible_asset::spec_supply(post_fa_addr)
    == (if (old(exists<fungible_asset::ConcurrentSupply>(post_fa_addr))) {
        old(fungible_asset::spec_supply(post_fa_addr))
    } else { 0 }) + amount;
